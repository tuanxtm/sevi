  target_compile_definitions(sevi-settings PRIVATE SEVI_VERSION="${SEVI_VERSION}" SEVI_DATADIR="${CMAKE_INSTALL_FULL_DATADIR}/sevi")
  target_link_libraries(sevi-settings PRIVATE PkgConfig::ADW)
  pkg_check_modules(GTKX11 IMPORTED_TARGET gtk4-x11 x11)

    target_compile_definitions(sevi-settings PRIVATE SEVI_HAVE_X11)
    target_link_libraries(sevi-settings PRIVATE PkgConfig::GTKX11)
  endif()

  install(FILES sevi.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES ${ROOT}/site/assets/icon.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps RENAME sevi.png)
