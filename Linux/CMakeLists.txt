# Sevi on Linux: the engine, the test tools (for Tools/autoswitch/check.sh) and the IBus engine.
# Configure:  cmake -S Linux -B Linux/build -G Ninja -DCMAKE_BUILD_TYPE=Release
# Build:      cmake --build Linux/build
cmake_minimum_required(VERSION 3.21)
project(Sevi LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Install to /usr like the packages do: the IBus component file bakes the libexec path in, so a /usr/local
# default would make the .deb/.rpm point at a binary that is not there.
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "/usr" CACHE PATH "install prefix" FORCE)
endif()
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/..)
set(ENGINE ${ROOT}/Engine)
set(AS ${ROOT}/Tools/autoswitch)
option(SEVI_IBUS "Build the IBus engine and native settings application" ON)
option(SEVI_RUST_DESKTOP "Build the complete Rust desktop host" ON)
set(ENGINE_SRC ${ENGINE}/Engine.cpp ${ENGINE}/Vietnamese.cpp ${ENGINE}/SmartSwitchKey.cpp)
option(SEVI_RUST_ENGINE "Use the Rust typing engine" ON)
if(SEVI_RUST_ENGINE)
  include(${ROOT}/cmake/RustEngine.cmake)
  set(ENGINE_SRC ${ENGINE}/RustBridge.cpp)
endif()
# One version for every platform: Cargo.toml's workspace version.
if(NOT SEVI_VERSION)
  file(STRINGS "${ROOT}/Cargo.toml" _mv REGEX "^version = ")
  string(REGEX REPLACE "^version = \"([^\"]+)\".*" "\\1" SEVI_VERSION "${_mv}")
endif()
message(STATUS "Sevi version ${SEVI_VERSION}")

if(SEVI_RUST_DESKTOP AND SEVI_IBUS AND SEVI_RUST_ENGINE)
  include(${ROOT}/Linux/RustDesktop.cmake)
else()
add_library(sevi_engine STATIC ${ENGINE_SRC} ${ENGINE}/Driver.cpp ${ENGINE}/TouchModel.cpp)
target_include_directories(sevi_engine PUBLIC ${ENGINE} ${ENGINE}/platforms)
target_compile_options(sevi_engine PUBLIC -w -O2 -fPIC)

add_library(sevi_engine_testing STATIC ${ENGINE_SRC})
target_include_directories(sevi_engine_testing PUBLIC ${ENGINE} ${ENGINE}/platforms)
target_compile_definitions(sevi_engine_testing PUBLIC AS_TESTING)
target_compile_options(sevi_engine_testing PUBLIC -w -O2)
if(SEVI_RUST_ENGINE)
  foreach(lib sevi_engine sevi_engine_testing)
    target_compile_definitions(${lib} PUBLIC SEVI_RUST_ENGINE)
    target_link_libraries(${lib} PUBLIC sevi_rust)
  endforeach()
endif()

function(sevi_tool name src outdir lib)
  add_executable(${name} ${src})
  target_link_libraries(${name} PRIVATE ${lib})
  set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tools")
endfunction()
sevi_tool(harness   ${AS}/test/harness.cpp     ${AS}/test   sevi_engine)
sevi_tool(sim       ${AS}/sim/sim.cpp           ${AS}/sim    sevi_engine_testing)
sevi_tool(drvlazy   ${AS}/driver/drvlazy.cpp    ${AS}/driver sevi_engine)
sevi_tool(bench     ${AS}/driver/bench.cpp      ${AS}/driver sevi_engine)
sevi_tool(undo_test ${AS}/driver/undo_test.cpp  ${AS}/driver sevi_engine)
sevi_tool(touchsim  ${AS}/touch/touchsim.cpp    ${AS}/touch  sevi_engine)

option(SEVI_IBUS "Build the IBus engine (needs libibus-1.0 + gtk4/libadwaita)" ON)
if(SEVI_IBUS AND EXISTS ${ROOT}/Linux/ibus/CMakeLists.txt)
  add_subdirectory(${ROOT}/Linux/ibus ibus)
endif()
endif()

# ---- packages: .deb (Ubuntu/Debian) and .rpm (Fedora) via CPack -----------------------------------------
# The Arch .pkg.tar.zst is repacked from the .deb in CI (Linux/packaging/make-pacman.sh), not a CPack generator.
set(CPACK_PACKAGE_NAME "sevi")
set(CPACK_PACKAGE_VERSION "${SEVI_VERSION}")
set(CPACK_PACKAGE_CONTACT "Tony Dinh <patent@tdinh.me>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Sevi - Vietnamese input (Telex) with automatic English/Vietnamese switching")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://vikey.app")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "ibus (>= 1.5), libpango-1.0-0, libpangocairo-1.0-0, libcairo2, libegl1, libgl1, libxkbcommon0, libwayland-client0, libx11-6, libgstreamer1.0-0, gstreamer1.0-plugins-base, gstreamer1.0-plugins-good, libnotify-bin, fonts-noto-color-emoji, curl")
if(NOT SEVI_RUST_DESKTOP)
  string(APPEND CPACK_DEBIAN_PACKAGE_DEPENDS ", libgtk-4-1, libadwaita-1-0")
endif()
set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "amd64")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "arm64")
endif()
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${ROOT}/Linux/packaging/postinst")
set(CPACK_RPM_PACKAGE_REQUIRES "ibus >= 1.5, pango, cairo, mesa-libEGL, mesa-libGL, libxkbcommon, wayland-libs, libX11, gstreamer1, gstreamer1-plugins-base, gstreamer1-plugins-good, libnotify, google-noto-emoji-color-fonts, curl")
if(NOT SEVI_RUST_DESKTOP)
  string(APPEND CPACK_RPM_PACKAGE_REQUIRES ", gtk4, libadwaita")
endif()
set(CPACK_RPM_PACKAGE_LICENSE "GPL-3.0")
set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
set(CPACK_RPM_POST_INSTALL_SCRIPT_FILE "${ROOT}/Linux/packaging/postinst")
set(CPACK_GENERATOR "DEB;RPM")
include(CPack)
